{{/* Version banner (#25). Two cases, both derived from params + the URL — no hard-coded version strings: - preview: a version that is documented but not released yet. It sits ABOVE latest, so the "older version" wording would be exactly backwards. - older: any other available version that is not latest. */}} {{ with site.Params.versions }} {{ $seg := index (split (strings.TrimPrefix "/" $.RelPermalink) "/") 0 }} {{ $cur := strings.TrimPrefix "v" $seg }} {{ $rest := strings.TrimPrefix (printf "/v%s" $cur) $.RelPermalink }} {{/* The same page may not exist in the other version (a page added in a preview tree, say). Fall back to that version's home rather than linking at a 404. */}} {{ $to := printf "/v%s%s" .latest $rest }} {{ if not (site.GetPage (strings.TrimSuffix "/" $to)) }}{{ $to = printf "/v%s/" .latest }}{{ end }} {{ if and .preview (eq $cur .preview) }}
You're reading the docs for v{{ $cur }}, which isn't released yet. Commands here pull :edge, the newest pre-release build. Go to the current release (v{{ .latest }}) →
{{ else if and (in .available $cur) (ne $cur .latest) }}
You're viewing v{{ $cur }} — an older version. Go to the latest (v{{ .latest }}) →
{{ end }} {{ end }}
